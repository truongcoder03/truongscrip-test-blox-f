local function cleanup()
	unloaded = true
	for _, conn in ipairs(connections) do
		if conn and conn.Connected then
			pcall(function() conn:Disconnect() end)
		end
	end
	for _, draw in ipairs(drawings) do
		if draw and draw.Remove then
			pcall(function() draw:Remove() end)
		end
	end
	for _, inst in ipairs(instances) do
		if inst and inst.Destroy then
			pcall(function() inst:Destroy() end)
		end
	end
	pcall(function()
		local char = LocalPlayer.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum then hum.WalkSpeed = 16 end
	end)
	table.clear(connections)
	table.clear(drawings)
	table.clear(instances)
end

local function getHeadshot(username: string): string?
	if not req then return nil end
	local ok, response = pcall(function()
		return req({
			Url = "https://users.roblox.com/v1/usernames/users",
			Method = "POST",
			Body = HttpService:JSONEncode({ usernames = { username }, excludeBannedUsers = true }),
			Headers = { ["Content-Type"] = "application/json" }
		})
	end)
	if ok and response and response.Success and response.Body then
		local decodeOk, data = pcall(function()
			return HttpService:JSONDecode(response.Body)
		end)
		if decodeOk and data and data.data and data.data[1] then
			local userId = data.data[1].id
			return "https://www.roblox.com/headshot-thumbnail/image?userId=" .. tostring(userId) .. "&width=150&height=150&format=png"
		end
	end
	return nil
end

local function makeDraggable(handle: GuiObject, target: GuiObject): () -> boolean
	local dragging = false
	local moved = false
	local dragStart, startPos

	trackConn(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			moved = false
			dragStart = input.Position
			startPos = target.Position
		end
	end))

	trackConn(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end))

	trackConn(UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			if delta.Magnitude > 6 then moved = true end
			target.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end))

	return function() return moved end
end

local function findTarget(rootPos: Vector3, range: number): Model?
	local enemies = workspace:FindFirstChild("Enemies")
	if not enemies then return nil end
	local best, bestDist = nil, range
	for _, mob in ipairs(enemies:GetChildren()) do
		local hum = mob:FindFirstChildOfClass("Humanoid")
		local r = mob:FindFirstChild("HumanoidRootPart")
		if hum and r and hum.Health > 0 then
			local d = (r.Position - rootPos).Magnitude
			if d < bestDist then
				best = mob
				bestDist = d
			end
		end
	end
	return best
end

local function acceptQuest()
	local data = LocalPlayer:FindFirstChild("Data")
	local lvlObj = data and data:FindFirstChild("Level")
	local level = (lvlObj and lvlObj.Value) or 1
	local chosen = questTable[1]
	for _, q in ipairs(questTable) do
		if level >= q[1] then chosen = q end
	end
	task.spawn(function()
		pcall(function()
			local remotes = ReplicatedStorage:FindFirstChild("Remotes")
			local commf = remotes and remotes:FindFirstChild("CommF_")
			if commf then
				cloneref(commf):InvokeServer("StartQuest", chosen[2], chosen[3])
			end
		end)
	end)
end

local attackCount = 0
trackConn(RunService.Heartbeat:Connect(function() end))

task.spawn(function()
	local noTargetTime = 0
	while not unloaded do
		task.wait(0.1)
		if not config.farm then
			noTargetTime = 0
			continue
		end
		local root = getRoot()
		local hum = getHum()
		if root and hum and hum.Health > 0 then
			local target = findTarget(root.Position, 500)
			if target then
				noTargetTime = 0
				local tr = target:FindFirstChild("HumanoidRootPart")
				if tr then
					pcall(function()
						root.CFrame = CFrame.lookAt(tr.Position + Vector3.new(0, config.farmHeight, 0), tr.Position)
						root.AssemblyLinearVelocity = Vector3.zero
					end)
					attackCount += 1
					if attackCount % 2 == 0 then
						pcall(function()
							local cam = workspace.CurrentCamera
							VirtualUser:Button1Down(Vector2.new(0, 0), cam.CFrame)
							VirtualUser:Button1Up(Vector2.new(0, 0), cam.CFrame)
						end)
					end
				end
			else
				noTargetTime += 0.1
				if config.autoQuest and noTargetTime >= 3 then
					noTargetTime = 0
					acceptQuest()
				end
			end
		end
	end
end)

local speedApplied = false
local upHeld = false
local downHeld = false

trackConn(RunService.Heartbeat:Connect(function()
	if unloaded then return end
	local char = getChar()
	local hum = getHum()
	local root = getRoot()
	if not hum or not root or hum.Health <= 0 then return end

	if config.speed then
		hum.WalkSpeed = config.speedValue
		speedApplied = true
	elseif speedApplied then
		hum.WalkSpeed = 16
		speedApplied = false
	end

	if config.jump then
		hum.UseJumpPower = true
		hum.JumpPower = config.jumpValue
	end

	if config.fly then
		local move = hum.MoveDirection
		local vert = 0
		if upHeld or UserInputService:IsKeyDown(Enum.KeyCode.Space) then vert += 1 end
		if downHeld or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vert -= 1 end
		local dir = move + Vector3.new(0, vert, 0)
		if dir.Magnitude > 1 then dir = dir.Unit end
		local yVel = (vert == 0) and 0 or (dir.Y * config.flySpeed)
		root.AssemblyLinearVelocity = Vector3.new(dir.X * config.flySpeed, yVel, dir.Z * config.flySpeed)
	end
end))

trackConn(RunService.Stepped:Connect(function()
	if unloaded or not config.noclip then return end
	local char = getChar()
	if not char then return end
	for _, p in ipairs(char:GetDescendants()) do
		if p:IsA("BasePart") and p.CanCollide then
			p.CanCollide = false
		end
	end
end))

task.spawn(function()
	while not unloaded do
		task.wait(0.5)
		local myRoot = getRoot()

		for key, data in pairs(espCache) do
			local stale = false
			if typeof(key) == "Instance" then
				stale = (not key.Parent) or (key:IsA("Player") and not key.Parent)
			end
			if stale then clearEsp(key) end
		end

		if config.esp then
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LocalPlayer and plr.Character then
					local char = plr.Character
					local root = char:FindFirstChild("HumanoidRootPart")
					local hum = char:FindFirstChildOfClass("Humanoid")
					if root and hum and hum.Health > 0 then
						if not espCache[plr] then
							local hl = Instance.new("Highlight")
							hl.Adornee = char
							hl.FillColor = ACCENT
							hl.OutlineColor = Color3.fromRGB(255, 255, 255)
							hl.FillTransparency = 0.6
							hl.Parent = espFolder
							local tag = makeTag(root, plr.DisplayName, TEXT)
							espCache[plr] = { hl, tag }
						end
						local dist = myRoot and math.floor((root.Position - myRoot.Position).Magnitude) or 0
						local tag = espCache[plr][2]
						local lbl = tag:FindFirstChildOfClass("TextLabel")
						if lbl then
							lbl.Text = plr.DisplayName .. " [" .. tostring(math.floor(hum.Health)) .. "] " .. dist .. "m"
						end
					end
				end
			end
		else
			for key, _ in pairs(espCache) do
				if typeof(key) == "Instance" and key:IsA("Player") then
					clearEsp(key)
				end
			end
		end

		if config.fruitEsp then
			for _, child in ipairs(workspace:GetChildren()) do
				if child:IsA("Tool") and string.find(string.lower(child.Name), "fruit") then
					local handle = child:FindFirstChild("Handle")
					if handle and not espCache[child] then
						local hl = Instance.new("Highlight")
						hl.Adornee = child
						hl.FillColor = Color3.fromRGB(80, 220, 120)
						hl.OutlineColor = Color3.fromRGB(255, 255, 255)
						hl.FillTransparency = 0.5
						hl.Parent = espFolder
						local tag = makeTag(handle, child.Name, Color3.fromRGB(120, 255, 160))
						espCache[child] = { hl, tag }
					end
				end
			end
			for key, _ in pairs(espCache) do
				if typeof(key) == "Instance" and key:IsA("Tool") and not key.Parent then
					clearEsp(key)
				end
			end
		else
			for key, _ in pairs(espCache) do
				if typeof(key) == "Instance" and key:IsA("Tool") then
					clearEsp(key)
				end
			end
		end
	end
end)

local function teleportTo(plr: Player)
	local char = plr.Character
	local tr = char and char:FindFirstChild("HumanoidRootPart")
	local root = getRoot()
	if tr and root then
		pcall(function()
			root.CFrame = tr.CFrame * CFrame.new(0, 0, 4)
		end)
	end
end

local function serverHop()
	if not req then
		notify("Executor không hỗ trợ request")
		return
	end
	task.spawn(function()
		local ok, res = pcall(function()
			return req({
				Url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", game.PlaceId),
				Method = "GET"
			})
		end)
		if ok and res and res.Success and res.Body then
			local decOk, data = pcall(function()
				return HttpService:JSONDecode(res.Body)
			end)
			if decOk and data and data.data then
				for _, s in ipairs(data.data) do
					if s.id ~= game.JobId and s.playing < s.maxPlayers then
						notify("Đang chuyển server...")
						pcall(function()
							TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
						end)
						return
					end
				end
			end
		end
		notify("Không tìm thấy server phù hợp")
	end)
end

local function rejoin()
	pcall(function()
		TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
	end)
end
