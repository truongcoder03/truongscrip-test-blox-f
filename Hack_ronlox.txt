local connections = {}

local function trackConn(conn: RBXScriptConnection): RBXScriptConnection
	table.insert(connections, conn)
	return conn
end

local function trackInst(inst: Instance): Instance
	table.insert(instances, inst)
	return inst
end

local function cleanup()
	for _, conn in ipairs(connections) do
		if conn and conn.Connected then
			pcall(function() conn:Disconnect() end)
		end
	end
	for _, draw in ipairs(drawings) do
		if draw and draw.Remove then
			pcall(function() draw:Remove() end)
		end
	end
	for _, inst in ipairs(instances) do
		if inst and inst.Destroy then
			pcall(function() inst:Destroy() end)
		end
	end
	table.clear(connections)
	table.clear(drawings)
	table.clear(instances)
end

local function setStatus(text: string)
	if statusLabel then
		statusLabel.Text = text
	end

local function getHum(): Humanoid?
	local c = LocalPlayer.Character
	return c and c:FindFirstChildOfClass("Humanoid") or nil
end

local function getHRP(): BasePart?
	local c = LocalPlayer.Character
	local hrp = c and c:FindFirstChild("HumanoidRootPart")
	if hrp and hrp:IsA("BasePart") then
		return hrp
	end
	return nil
end

local function isAlive(): boolean
	local h = getHum()
	return h ~= nil and h.Health > 0
end

local function getStat(name: string): number
	local data = LocalPlayer:FindFirstChild("Data")
	local v = data and data:FindFirstChild(name)
	if v and (v:IsA("IntValue") or v:IsA("NumberValue")) then
		return v.Value
	end
	return 0
end

local function getLevel(): number
	local lv = getStat("Level")
	return lv > 0 and lv or 1
end

local function invokeCommF(...: any): any
	local ok, res = pcall(function()
		local remotes = ReplicatedStorage:FindFirstChild("Remotes")
		local f = remotes and remotes:FindFirstChild("CommF_")
		if f and f:IsA("RemoteFunction") then
			return f:InvokeServer(...)
		end
		return nil
	end)
	if ok then
		return res
	end
	return nil
end

local function getQuestData(): QuestEntry
	local lv = getLevel()
	for _, q in ipairs(QuestTable) do
		if lv >= q.MinLv and lv <= q.MaxLv then
			return q
		end
	end
	return QuestTable[#QuestTable]
end

local function questActive(q: QuestEntry): boolean
	local main = PlayerGui:FindFirstChild("Main")
	local questFrame = main and main:FindFirstChild("Quest")
	if not questFrame or not questFrame:IsA("GuiObject") or not questFrame.Visible then
		return false
	end
	for _, d in ipairs(questFrame:GetDescendants()) do
		if d:IsA("TextLabel") and string.find(d.Text, q.Mob, 1, true) then
			return true
		end
	end
	return false
end

local function findPivotByName(name: string): CFrame?
	local direct = Workspace:FindFirstChild(name, true)
	if direct and (direct:IsA("Model") or direct:IsA("BasePart")) then
		local ok, pivot = pcall(function() return direct:GetPivot() end)
		if ok and pivot then
			return pivot
		end
	end
	for _, d in ipairs(Workspace:GetDescendants()) do
		if string.find(string.lower(d.Name), string.lower(name), 1, true) and (d:IsA("Model") or d:IsA("BasePart")) then
			local ok, pivot = pcall(function() return d:GetPivot() end)
			if ok and pivot then
				return pivot
			end
		end
	end
	return nil
end

local function flyTo(target: Vector3, cond: () -> boolean): boolean
	local started = os.clock()
	while cond() do
		if os.clock() - started > 40 then
			return false
		end
		local hrp = getHRP()
		local hum = getHum()
		if not hrp or not hum or hum.Health <= 0 then
			return false
		end
		local offset = target - hrp.Position
		local dist = offset.Magnitude
		if dist <= 3 then
			hrp.AssemblyLinearVelocity = Vector3.zero
			return true
		end
		local dt = RunService.Heartbeat:Wait()
		local step = math.min(dist, Config.FlySpeed * dt)
		local newPos = hrp.Position + offset.Unit * step
		hrp.CFrame = CFrame.new(newPos) * (hrp.CFrame - hrp.CFrame.Position)
		hrp.AssemblyLinearVelocity = Vector3.zero
	end
	return false
end

local function attack()
	local char = LocalPlayer.Character
	local hum = getHum()
	if not char or not hum then
		return
	end
	local tool = char:FindFirstChildOfClass("Tool")
	if not tool then
		local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
		local t = bp and bp:FindFirstChildOfClass("Tool")
		if t then
			pcall(function() hum:EquipTool(t) end)
		end
		return
	end
	pcall(function() tool:Activate() end)
	pcall(function()
		local cam = Workspace.CurrentCamera
		local camCF = cam and cam.CFrame or CFrame.new()
		VirtualUser:Button1Down(Vector2.new(0, 0), camCF)
		VirtualUser:Button1Up(Vector2.new(0, 0), camCF)
	end)
end

local function findNearestMob(name: string): Model?
	local hrp = getHRP()
	local folder = Workspace:FindFirstChild("Enemies")
	if not hrp or not folder then
		return nil
	end
	local best: Model? = nil
	local bestDist = math.huge
	for _, m in ipairs(folder:GetChildren()) do
		if m:IsA("Model") and string.find(m.Name, name, 1, true) then
			local h = m:FindFirstChildOfClass("Humanoid")
			local mh = m:FindFirstChild("HumanoidRootPart")
			if h and mh and mh:IsA("BasePart") and h.Health > 0 then
				local d = (mh.Position - hrp.Position).Magnitude
				if d < bestDist then
					best = m
					bestDist = d
				end
			end
		end
	end
	return best
end

local function ensureQuest(alive: () -> boolean)
	local q = getQuestData()
	if questActive(q) then
		return
	end
	if os.clock() - lastQuestAttempt < 8 then
		return
	end
	lastQuestAttempt = os.clock()
	local npc = findPivotByName(q.Npc)
	if not npc then
		setStatus("Không tìm thấy NPC: " .. q.Npc)
		return
	end
	if not flyTo(npc.Position + Vector3.new(0, 3, 4), alive) then
		return
	end
	invokeCommF("AbandonQuest")
	invokeCommF("StartQuest", q.Quest, q.QLv)
	task.wait(1)
end

local function live(sessionId: number, key: string): () -> boolean
	return function()
		return Config.Session == sessionId and Config[key] == true
	end
end

local function farmStep(alive: () -> boolean)
	if not isAlive() then
		task.wait(0.5)
		return
	end
	if Config.AutoQuest then
		ensureQuest(alive)
	end
	local q = getQuestData()
	local mob = findNearestMob(q.Mob)
	if not mob then
		setStatus("Đang tìm " .. q.Mob .. "...")
		task.wait(0.5)
		return
	end
	local mhrp = mob:FindFirstChild("HumanoidRootPart")
	local mhum = mob:FindFirstChildOfClass("Humanoid")
	if not mhrp or not mhum or not mhrp:IsA("BasePart") then
		return
	end
	local hrp = getHRP()
	if hrp and (hrp.Position - mhrp.Position).Magnitude > 150 then
		flyTo(mhrp.Position + Vector3.new(0, Config.AttackHeight, 0), alive)
	end
	local deadline = os.clock() + 60
	while alive() and os.clock() < deadline and mob.Parent and mhum.Health > 0 and isAlive() do
		if Config.AutoQuest and not questActive(q) then
			ensureQuest(alive)
		end
		local myHrp = getHRP()
		local tHrp = mob:FindFirstChild("HumanoidRootPart")
		if not myHrp or not tHrp or not tHrp:IsA("BasePart") then
			break
		end
		setStatus("Farm: " .. mob.Name .. " | Lv " .. tostring(getLevel()))
		myHrp.CFrame = CFrame.lookAt(tHrp.Position + Vector3.new(0, Config.AttackHeight, 0), tHrp.Position)
		myHrp.AssemblyLinearVelocity = Vector3.zero
		attack()
		task.wait(Config.AttackDelay)
	end
end

local function runLoop(s: number, key: string, step: (() -> ()) -> ())
	local alive = live(s, key)
	while alive() do
		local ok, err = pcall(step, alive)
		if not ok then
			setStatus("Lỗi: " .. tostring(err))
			task.wait(1)
		else
			task.wait(0.1)
		end
	end
end

local function farmLoop(s: number)
	runLoop(s, "AutoFarm", farmStep)
	setStatus("Đã tắt auto farm")
end

local function collectChests(alive: () -> boolean)
	local tried: {[Instance]: boolean} = {}
	for _, obj in ipairs(Workspace:GetDescendants()) do
		if not alive() then
			return
		end
		if (obj:IsA("BasePart") or obj:IsA("Model")) and not tried[obj] and obj.Parent then
			if string.find(string.lower(obj.Name), "chest", 1, true) then
				tried[obj] = true
				local ok, pivot = pcall(function() return obj:GetPivot() end)
				if ok and pivot then
					setStatus("Đang tới rương: " .. obj.Name)
					local arrived = flyTo(pivot.Position + Vector3.new(0, 3, 0), alive)
					if arrived then
						local part: BasePart? = nil
						if obj:IsA("BasePart") then
							part = obj
						else
							part = obj:FindFirstChildWhichIsA("BasePart", true)
						end
						local h = getHRP()
						if part and h and firetouchinterest then
							pcall(function()
								firetouchinterest(h, part, 0)
								task.wait()
								firetouchinterest(h, part, 1)
							end)
						end
						task.wait(0.4)
					end
				end
			end
		end
	end
end

local function chestLoop(s: number)
	runLoop(s, "AutoChest", function(alive)
		if isAlive() then
			collectChests(alive)
		end
		task.wait(2)
	end)
	setStatus("Đã tắt auto chest")
end

local function startAll()
	local s = Config.Session
	if Config.AutoFarm then
		task.spawn(farmLoop, s)
	end
	if Config.AutoChest then
		task.spawn(chestLoop, s)
	end
end

local function setFeature(key: string, value: boolean)
	Config[key] = value
	Config.Session += 1
	startAll()
	for _, r in ipairs(refreshers) do
		r()
	end
end

local function travelTo(name: string)
	Config.Session += 1
	local s = Config.Session
	local cf = findPivotByName(name)
	if not cf then
		setStatus("Không tìm thấy: " .. name)
		return
	end
	Config.Traveling = true
	setStatus("Đang bay tới " .. name)
	task.spawn(function()
		local ok = flyTo(cf.Position + Vector3.new(0, 20, 0), function()
			return Config.Session == s
		end)
		Config.Traveling = false
		setStatus(ok and ("Đã tới " .. name) or "Dịch chuyển bị hủy")
	end)
end

local function makeDraggable(handle: GuiObject, target: GuiObject)
	local dragging = false
	local dragStart = Vector2.zero
	local startPos = target.Position
	trackConn(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = Vector2.new(input.Position.X, input.Position.Y)
			startPos = target.Position
		end
	end))
	trackConn(UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
			target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end))
	trackConn(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end))
end

local function applyMove()
	local hum = getHum()
	if hum and moveState.active then
		hum:Move(moveState.vec, true)
	end
end

local function stopMove()
	moveState.active = false
	moveState.input = nil
	moveState.vec = Vector3.zero
	local hum = getHum()
	if hum then
		hum:Move(Vector3.zero, true)
	end
end

trackConn(movePad.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
		moveState.active = true
		moveState.input = input
		moveState.origin = Vector2.new(input.Position.X, input.Position.Y)
		moveState.vec = Vector3.zero
	end
end))

trackConn(UserInputService.InputChanged:Connect(function(input)
	if not moveState.active or not moveState.input then
		return
	end
	if input.UserInputType ~= Enum.UserInputType.Touch and input.UserInputType ~= Enum.UserInputType.MouseMovement then
		return
	end
	if moveState.input.UserInputType == Enum.UserInputType.Touch and input ~= moveState.input then
		return
	end
	local delta = Vector2.new(input.Position.X, input.Position.Y) - moveState.origin
	local v = Vector2.new(delta.X / JOY_RADIUS, -delta.Y / JOY_RADIUS)
	if v.Magnitude > 1 then
		v = v.Unit
	end
	if v.Magnitude < JOY_DEADZONE then
		v = Vector2.zero
	end
	moveState.vec = Vector3.new(v.X, 0, -v.Y)
	applyMove()
end))

trackConn(UserInputService.InputEnded:Connect(function(input)
	if not moveState.active or not moveState.input then
		return
	end
	local isEnd = input == moveState.input
	if moveState.input.UserInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseButton1 then
		isEnd = true
	end
	if isEnd then
		stopMove()
	end
end))

		hum:ChangeState(Enum.HumanoidStateType.Jumping)
	end
end

	if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
		doJump()
	end
end))

local function createToggle(text: string, key: string, order: number, isHud: boolean?)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, -4, 0, 46)
	btn.BackgroundColor3 = PANEL
	btn.Font = Enum.Font.GothamBold
	btn.TextSize = 15
	btn.TextColor3 = TEXT
	btn.LayoutOrder = order
	btn.Parent = body
	addCorner(btn, 10)
	addStroke(btn, LAVENDER, 1)
	local function refresh()
		local on = Config[key] == true
		btn.Text = text .. (on and ": ON" or ": OFF")
		btn.BackgroundColor3 = on and PINK or PANEL
	end
	refresh()
	table.insert(refreshers, refresh)
	trackConn(btn.Activated:Connect(function()
		if isHud then
			Config[key] = not (Config[key] == true)
			applyGameHUD()
			for _, r in ipairs(refreshers) do
				r()
			end
		else
			setFeature(key, not (Config[key] == true))
		end
	end))
end

local function createStepper(label: string, key: string, step: number, minV: number, maxV: number, order: number)
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1, -4, 0, 46)
	row.BackgroundColor3 = PANEL
	row.LayoutOrder = order
	row.Parent = body
	addCorner(row, 10)

	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, -140, 1, 0)
	lbl.Position = UDim2.new(0, 12, 0, 0)
	lbl.BackgroundTransparency = 1
	lbl.Text = label
	lbl.Font = Enum.Font.GothamMedium
	lbl.TextSize = 14
	lbl.TextColor3 = TEXT
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Parent = row

	local minus = Instance.new("TextButton")
	minus.Size = UDim2.fromOffset(40, 34)
	minus.Position = UDim2.new(1, -130, 0, 6)
	minus.BackgroundColor3 = BG
	minus.Text = "-"
	minus.Font = Enum.Font.GothamBold
	minus.TextSize = 18
	minus.TextColor3 = TEXT
	minus.Parent = row
	addCorner(minus, 8)

	local val = Instance.new("TextLabel")
	val.Size = UDim2.fromOffset(44, 34)
	val.Position = UDim2.new(1, -86, 0, 6)
	val.BackgroundTransparency = 1
	val.Font = Enum.Font.GothamBold
	val.TextSize = 14
	val.TextColor3 = PINK
	val.Parent = row

	local plus = Instance.new("TextButton")
	plus.Size = UDim2.fromOffset(40, 34)
	plus.Position = UDim2.new(1, -42, 0, 6)
	plus.BackgroundColor3 = BG
	plus.Text = "+"
	plus.Font = Enum.Font.GothamBold
	plus.TextSize = 18
	plus.TextColor3 = TEXT
	plus.Parent = row
	addCorner(plus, 8)

	local function refresh()
		val.Text = tostring(Config[key])
	end
	refresh()
	table.insert(refreshers, refresh)

	trackConn(minus.Activated:Connect(function()
		Config[key] = math.clamp(Config[key] - step, minV, maxV)
		refresh()
	end))
	trackConn(plus.Activated:Connect(function()
		Config[key] = math.clamp(Config[key] + step, minV, maxV)
		refresh()
	end))
end

	for _, key in ipairs({"AutoFarm", "AutoQuest", "AutoChest", "Traveling"}) do

task.spawn(function()
	while screen.Parent do
		local hum = getHum()
		if hum then
			local maxHp = math.max(hum.MaxHealth, 1)
			hpFill.Size = UDim2.fromScale(math.clamp(hum.Health / maxHp, 0, 1), 1)
		else
			hpFill.Size = UDim2.fromScale(0, 1)
		end
		hudName.Text = LocalPlayer.Name
		hudInfo.Text = "Lv " .. tostring(getLevel()) .. " | Beli " .. tostring(getStat("Beli"))
		if not (Config.AutoFarm or Config.AutoChest) then
			setStatus("Level " .. tostring(getLevel()) .. " | Quest: " .. getQuestData().Quest)
		end
		task.wait(0.25)
	end
end)
